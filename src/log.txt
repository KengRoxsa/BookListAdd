##
ยังไม่ค่อยเก็ตท่า {child} ของ context เท่าไหร่ มันเหมือนการรับ มากกว่าส่ง prop แต่ใีนใช้ได้ 
function Provider({children}) 

object ไม่สามารถ render ใน JSX ตรงๆ ได้ ปัญหานี้อาจเจอ ตอนใช้ทั้ง bookprovider , bookcontext 
ไม่แน่ใจ แต่มันคือ obj render in jsx ไม่ได้
ex 
const value = useContext(BooksContext);

return (
  <div>
    {value}  // ❌ Error เพราะ value เป็น object
  </div>
);
 == แตก

 const { count } = useContext(BooksContext);

return (
  <div>
    Count: {count}  // ✅ ใช้ค่าที่เป็น number/string ได้
  </div>
);
